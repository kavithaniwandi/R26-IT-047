@asynccontextmanager
async def lifespan(app: FastAPI) -> AsyncGenerator[None, None]:
    from app.models import appeal_guidance_service
    appeal_guidance_service.GAP_CHECKS = appeal_guidance_service.load_gap_statistics()
    # ------------------------------------------------------------------ #
    # Venv / import guard — logs ERROR immediately if the wrong venv is   #
    # active so the missing-module root cause is obvious on startup.       #
    # ------------------------------------------------------------------ #
    _required = {
        "google.genai":    "google-genai  (Gemini generation)",
        "huggingface_hub": "huggingface_hub (HF fallback)",
    }
    for _mod, _desc in _required.items():
        try:
            importlib.import_module(_mod)
        except ImportError as _exc:
            _startup_log.error(
                "MISSING IMPORT '%s' (%s) in Python %s. "
                "Run the server from backend/venv, not .venv-sev. Detail: %s",
                _mod, _desc, sys.executable, _exc,
            )

    init_db()
    try:
        from app.models.quality_service import warm_up

        warm_up()
    except Exception as exc:
        _startup_log.error(
            "Quality model warmup failed: %s. Appeal scoring will return errors until fixed.",
            exc,
        )
    try:
        from app.services.mongo_service import connect_mongo

        await connect_mongo()
    except Exception as exc:
        _startup_log.error("MongoDB connection failed: %s", exc)
    try:
        yield
    finally:
        from app.services.mongo_service import close_mongo

        await close_mongo()

@app.post("/api/quality/evaluate", response_model=QualityScoreResponse)
async def evaluate_quality(request: QualityScoreRequest) -> QualityScoreResponse:
    """Evaluate donation appeal quality using the trained local ML model."""
    try:
        from app.models.quality_service import get_quality_score

        result = get_quality_score(request.appeal_text, language=request.language)
        from app.models.appeal_guidance_service import guidance
        metadata = await asyncio.to_thread(guidance, request.appeal_text, request.language, result,
                                         request.length_category)
        return QualityScoreResponse(
            **metadata,
            score=result["score"],
            raw_quality_score=result.get("raw_quality_score", result["score"]),
            status=result["status"],
            final_label=result["status"],
            predicted_class=result.get("predicted_class"),
            class_probs=result.get("class_probs", {}),
            method=result["method"],
            confidence=result["confidence"],
            confidence_normalised=result["confidence_normalised"],
            confidence_display=result["confidence_display"],
            low_confidence=result["low_confidence"],
        )
    except QualityModelLoadError as exc:
        return JSONResponse(
            status_code=503, content={"status": "failed", "detail": str(exc)}
        )
    except ValueError as exc:
        raise HTTPException(status_code=422, detail=str(exc)) from exc
    except Exception as exc:
        raise HTTPException(
            status_code=500,
            detail="Quality evaluation failed.",
        ) from exc


def _appeal_client(request):
    # Forwarded headers are intentionally ignored: only trust the ASGI peer IP.
    return request.client.host if request is not None and request.client else "local"


def _appeal_response(variants, request, payload=None):
    from app.models.appeal_guidance_service import recommended
    from app.services.appeal_budget import improvement_uses, session
    all_variants = variants
    if payload is not None:
        state = session(payload, _appeal_client(request))
        all_variants = [v for group in state["results"].values() for v in group] or variants
    partial = any(v.get("partial", False) for v in all_variants)
    calls = max((v.get("gemini_calls_used", 0) for v in all_variants), default=0)
    if payload is not None:
        partial = partial or state.get("partial", False)
        calls = state["budget"].calls + state["remaining_budget"].calls
    return GenerateAppealVariantsResponse(
        variants=variants, recommended=recommended(all_variants, (payload or {}).get("length_category", "short")),
        remaining_improve_uses=improvement_uses(_appeal_client(request))[1],
        gemini_calls_used=calls, partial=partial)

@app.post("/api/quality/analyse")
async def analyse_appeal(payload: dict, http_request: Request = None) -> dict:
    """Pure ML quality evaluation plus structured issue diagnosis."""
    appeal_text = (payload.get("appeal_text") or "").strip()
    language = payload.get("language") or "English"

    if not appeal_text:
        raise HTTPException(status_code=422, detail="appeal_text is required.")

    try:
        from app.models.gemini_service import _diagnose_weaknesses
        from app.models.quality_service import get_quality_score

        quality = get_quality_score(appeal_text, language=language)
        from app.models.appeal_guidance_service import guidance
        from app.services.appeal_budget import improvement_uses
        metadata = await asyncio.to_thread(guidance, appeal_text, language, quality, None)
        result = {
            **metadata,
            "original_appeal": payload.get("appeal_text", ""),
            "remaining_improve_uses": improvement_uses(_appeal_client(http_request))[1],
            "score": quality["score"],
            "raw_quality_score": quality.get("raw_quality_score", quality["score"]),
            "label": quality["status"],
            "final_label": quality["status"],
            "predicted_class": quality.get("predicted_class"),
            "class_probs": quality.get("class_probs", {}),
            "confidence": quality["confidence"],
            "low_confidence": quality["low_confidence"],
            "method": quality["method"],
            "issues": [],
        }
        from app.services.mongo_service import log_appeal_analysis

        asyncio.create_task(
            log_appeal_analysis(appeal_text, language, result, source="analyser")
        )
        return result
    except QualityModelLoadError as exc:
        return JSONResponse(
            status_code=503, content={"status": "failed", "detail": str(exc)}
        )
    except ValueError as exc:
        raise HTTPException(status_code=422, detail=str(exc)) from exc
    except Exception as exc:
        raise HTTPException(status_code=500, detail="Appeal analysis failed.") from exc


_APPEAL_LOG_TASKS: set[asyncio.Task] = set()


def _schedule_appeal_log(log_function, *args, **kwargs) -> None:
    async def guarded_log():
        try:
            await log_function(*args, **kwargs)
        except Exception as exc:
            logger.warning("MongoDB logging failed (non-fatal): %s", type(exc).__name__)

    task = asyncio.create_task(guarded_log())
    _APPEAL_LOG_TASKS.add(task)
    task.add_done_callback(_APPEAL_LOG_TASKS.discard)


@app.post("/api/quality/improve")
async def improve_appeal(payload: dict, http_request: Request = None) -> dict:
    """Generate an improved donation appeal after ML issue diagnosis."""
    appeal_text = payload.get("appeal_text") or ""
    language = payload.get("language") or "English"

    if not appeal_text.strip():
        raise HTTPException(status_code=422, detail="appeal_text is required.")

    try:
        from app.models.appeal_generation_service import repair
        result = await repair(appeal_text, {**payload, "language": language}, _appeal_client(http_request))
        from app.services.mongo_service import log_appeal_analysis

        _schedule_appeal_log(
            log_appeal_analysis, appeal_text,
            language,
            result,
            event_type="improvement",
        )
        return result
    except QualityModelLoadError as exc:
        return JSONResponse(
            status_code=503, content={"status": "failed", "detail": str(exc)}
        )
    except Exception as exc:
        raise HTTPException(status_code=500, detail="Appeal improvement failed.") from exc


@app.post("/api/generate-appeal", response_model=GenerateAppealResponse)
async def generate_appeal(request: GenerateAppealRequest) -> GenerateAppealResponse:
    """Generate a donation appeal from campaign details."""
    try:
        from app.models.gemini_service import generate_donation_appeal

        campaign_data = request.model_dump()
        appeal_text = await asyncio.to_thread(generate_donation_appeal, campaign_data)
        try:
            from app.services.mongo_service import log_appeal_generation
            _schedule_appeal_log(
                log_appeal_generation, campaign_data,
                [
                    {
                        "appeal_text": appeal_text,
                        "quality_score": None,
                        "quality_label": None,
                        "provider": None,
                        "style": "single",
                    }
                ],
            )
        except Exception as log_exc:
            logger.warning("MongoDB logging failed (non-fatal): %s", type(log_exc).__name__)
        return GenerateAppealResponse(appeal_text=appeal_text)
    except QualityModelLoadError as exc:
        return JSONResponse(
            status_code=503, content={"status": "failed", "detail": str(exc)}
        )
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc)) from exc
    except Exception as exc:
        raise HTTPException(status_code=500, detail="Appeal generation failed.") from exc

REJECTION_LOG_PATH = Path(tempfile.gettempdir()) / "appeal_rejected_rewrites.log"


def _log_rejected_rewrite(text, reason, failed_guards=None):
    """Keep guard diagnostics locally; redact configured secrets before writing."""
    secrets = [str(value) for name, value in {**os.environ, **settings.model_dump()}.items()
               if value and any(token in name.upper() for token in ("KEY", "SECRET", "TOKEN", "PASSWORD", "URI"))]
    # Include runtime key providers (tests and deployments may inject them separately).
    from app.models import gemini_service as gs
    secrets.extend(gs._gemini_api_keys())
    redacted = text
    for secret in sorted(set(secrets), key=len, reverse=True):
        redacted = redacted.replace(secret, "[REDACTED]")
    redacted = re.sub(r"AIza[\w-]{20,}", "[REDACTED]", redacted)
    redacted = re.sub(r"(?i)([?&](?:key|api_key)=)[^\s&]+", r"\1[REDACTED]", redacted)
    record = {"time_utc": datetime.now(timezone.utc).isoformat(),
              "guard": reason, "failed_guards": failed_guards or [], "rewrite": redacted}
    try:
        with budgets.LOCK, REJECTION_LOG_PATH.open("a", encoding="utf-8") as stream:
            stream.write(json.dumps(record, ensure_ascii=False) + "\n")
    except OSError:
        logger.warning("Could not write local rejected-rewrite diagnostic")
    logger.info("Appeal rewrite rejected reason=%s diagnostic_log=%s", reason, REJECTION_LOG_PATH)


def _safe_rewrite(original, rewrite):
    """Conservative lexical admission gate; reject unsupported content rather than infer facts."""
    supplied = original
    words = lambda text: set(re.findall(r"[^\W\d_]+", text.casefold(), re.UNICODE))
    neutral = set("a an the and or but for to of in on at by with from into through this that these those your our their its is are was were be been being as it they we you us them please donate donation donations support supporting help helping provide providing give giving contribute contributions together community effort request requested need needed needs affected families people verified stated items can may will would should today".split())
    # 'today' is a deadline/urgency fact, not a neutral connective.
    neutral -= {"today", "families", "people", "affected", "community", "verified"}
    if not words(rewrite) <= words(supplied) | neutral:
        return False
    if set(re.findall(r"\d[\d,.]*", rewrite)) - set(re.findall(r"\d[\d,.]*", supplied)):
        return False
    for item in re.findall(r"\d[\d,.]*\s+[^\W\d_]+", rewrite.casefold()):
        if item not in supplied.casefold():
            return False
    for token in ("not", "never", "no", "without"):
        if bool(re.search(r"\b" + token + r"\b", original, re.I)) != bool(re.search(r"\b" + token + r"\b", rewrite, re.I)):
            return False
    return bool(rewrite.strip())

    from app.services.credit_monitor import is_exhausted
    configured = [getattr(settings, "GEMINI_API_KEY" + (str(i) if i > 1 else "")) for i in range(1, 12)]
    indexed = [(next((slot for slot, value in enumerate(configured, 1) if value == key), fallback), key)
               for fallback, key in enumerate(keys, 1)]
    selected = next(((i, key) for i, key in indexed if not is_exhausted(f"gemini_key_{i}")), None)
    if selected is None:
        budget.stopped = True
        raise budgets.BudgetExceeded("No usable Gemini key; using template")
    index, key = selected
    models = list(dict.fromkeys([settings.GEMINI_MODEL, *[m.strip() for m in settings.GEMINI_FALLBACK_MODELS.split(',') if m.strip()]]))
    for model_index, model in enumerate(models):
        for attempt in range(2 if model_index == 0 else 1):
            budget.consume()
            budget.last_model = model
            started = time.monotonic()
            token = budgets.CURRENT.set(budget)
            status, billable = "error", False
            try:
                kwargs = {} if model_index == 0 else {"model": model}
                result = await asyncio.wait_for(gs.call_gemini_with_key(prompt, 0.5, key, **kwargs),
                                                timeout=max(0.001, budget.remaining_seconds()))
                status, billable = "200", True
                return result, model
            except Exception as exc:
                message = str(exc).lower()
                code = getattr(exc, "code", None) or getattr(exc, "status_code", None)
                if code is None:
                    match = re.search(r"\b(400|401|403|404|429|500|503|504)\b", message)
                    code = match.group(1) if match else None
                status = str(code or type(exc).__name__)
                quota = any(word in message for word in ("quota", "resource_exhausted"))
                daily = status == "429" and bool(gs._DAILY_QUOTA_RE.search(message))
                # Read timeout may follow a dispatched/billable request; connect failures do not.
                billable = status == "429" or quota or isinstance(exc, asyncio.TimeoutError) or type(exc).__name__ == "ReadTimeout"
                if status in ("400", "401", "403") or daily or (quota and status != "429"):
                    budget.stop_flow()
                    raise budgets.BudgetExceeded(f"Provider rejected the call ({status}); generation stopped") from None
                hint = gs._retry_delay_seconds(exc, message) if status == "429" else 0.0
                retry_delay = (hint if status == "429" and 0 < hint <= 10 else
                               1.0 if status == "503" else 0.0)
                if attempt == 0 and model_index == 0 and retry_delay and retry_delay < budget.remaining_seconds():
                    await asyncio.sleep(retry_delay)
                    continue
                if status == "429":
                    raise budgets.BudgetExceeded("429 rate limit; no further retry within this endpoint budget") from None
                if status not in ("404", "500", "503", "504") and type(exc).__name__ not in ("ConnectionError", "ConnectTimeout"):
                    raise budgets.BudgetExceeded(f"Provider unavailable ({status})") from None
                break
            finally:
                budget.account(billable)
                budget.last_status = status
                budgets.CURRENT.reset(token)
                logger.info("Appeal Gemini key_index=%d model=%s status=%s elapsed_ms=%.1f calls_used=%d provider_attempts=%d",
                            index, model, status, (time.monotonic()-started)*1000, budget.calls, budget.flow.attempts)
    retries = 1 if len(models) == 1 and budget.attempts > 1 else 0
    cause = f"{budget.last_status} overloaded after {retries} retries" if budget.last_status == "503" else f"Provider unavailable ({budget.last_status})"
    raise budgets.BudgetExceeded(cause)


def _prompt(payload, remaining=False):
    from app.models import gemini_service as gs
    minimum, maximum = guide.length_band(payload.get("length_category", "short"))
    language = guide.quality._normalise_language(payload.get("language"))
    payload = {**payload, "language": language}
    prompt = gs._build_generation_prompt(payload)
    band = guide.length_config(payload.get("length_category", "short"))
    target_min, target_max = band["target_minimum"], band["target_maximum"]
    prompt += (f"\nTARGET: {target_min}-{target_max} words per appeal; aim comfortably above the minimum. "
               f"\nMANDATORY LENGTH: {minimum}–{maximum} whitespace-delimited words PER appeal. "
               f"Write at least {minimum} words; maximum {maximum}. Count before responding. "
               "Use only supplied facts. Do not invent deadlines, contacts, organizations, amounts, people or urgency. "
               "Length requirements override conflicting channel guidance.\n")
    if remaining:
        prompt += 'Return ONLY JSON: {"appeals":[{"style":"community","appeal_text":"..."},{"style":"urgent","appeal_text":"..."}]}. Exactly two different appeals; no markdown.'
    return prompt


def _parse(text):
    text = re.sub(r"^```(?:json)?\s*|\s*```$", "", text.strip())
    try:
        value = json.loads(text)
    except (ValueError, TypeError):
        return [], True
    items = value.get("appeals", []) if isinstance(value, dict) else value
    if not isinstance(items, list):
        return [], True
    valid, seen = [], set()
    for item in items:
        if not isinstance(item, dict):
            continue
        style, appeal = item.get("style"), item.get("appeal_text")
        if style in ("community", "urgent") and isinstance(appeal, str) and len(appeal.strip()) >= 10 and style not in seen:
            valid.append((style, appeal.strip()))
            seen.add(style)
    return valid, len(valid) != 2


async def generate(payload, remaining=False, client="local"):
    from app.models import gemini_service as gs
    gs._ensure_required_generation_fields(payload)
    state = budgets.session(payload, client)
    # A per-session lock prevents concurrent clicks from duplicating billed calls.
    lock = state.setdefault("lock", asyncio.Lock())
    async with lock:
        stage = "remaining" if remaining else "first"
        if stage in state["results"]:
            return state["results"][stage]
        budget = state["remaining_budget"] if remaining else state["budget"]
        # Network time budget covers active work across the progressive pair, not user idle time.
        spent = 0.0
        budget.started = time.monotonic() - spent
        started = time.monotonic()
        generated, model, error = [], "fallback_template", None
        try:
            text, model = await _call(_prompt(payload, remaining), budget)
            if remaining:
                generated, malformed = _parse(text)
                if malformed:
                    retry_text, retry_model = await _call(_prompt(payload, True) + "\nCorrect the JSON schema and return both styles.", budget, retry=True)
                    retry_valid, malformed = _parse(retry_text)
                    combined = {style: (appeal, model) for style, appeal in generated}
                    combined.update({style: (appeal, retry_model) for style, appeal in retry_valid})
                    generated = [(style, item[0], item[1]) for style, item in combined.items()]
                    if malformed:
                        error = "Provider returned incomplete or malformed two-style JSON after one retry"
                else:
                    generated = [(s, t, model) for s, t in generated]
            elif text.strip():
                generated = [("formal", text.strip(), model)]
            else:
                error = "Provider returned an empty appeal; using template"
        except (budgets.BudgetExceeded, asyncio.TimeoutError) as exc:
            error = str(exc) if isinstance(exc, budgets.BudgetExceeded) else "Generation time budget reached"
            generated = [(s, t, model) for s, t in generated if isinstance(t, str)]
        generation_ms = round((time.monotonic() - started) * 1000)
        state["spent_seconds"] = spent + time.monotonic() - started
        fallback = False
        if not generated and remaining:
            # Even an empty later response must not hide an unavailable quality model.
            await asyncio.to_thread(guide.quality._load_artifacts)
            state["results"][stage] = []
            state["partial"] = True
            logger.info("Appeal remaining partial=true variants=0 calls_used=%d provider_attempts=%d reason=%s", budget.calls, budget.flow.attempts, error)
            return []
        if not generated:
            generated = [("community" if remaining else "formal", gs._build_fallback_appeal(payload), "fallback_template")]
            fallback = True
        language = payload.get("language", "English")
        score_started = time.monotonic()
        if remaining:
            scores = await asyncio.to_thread(guide.batch_score, [item[1] for item in generated], language)
        else:
            scores = [await asyncio.to_thread(gs.score_text, generated[0][1], language)]
            scores[0]["final_label"] = scores[0].get("final_label", scores[0]["quality_label"])
        scoring_ms = round((time.monotonic() - score_started) * 1000, 1)
        variants = []
        all_metadata = guide.guidance_many([item[1] for item in generated], language, scores,
                                          payload.get("length_category", "short"))
        for (style, text, used_model), score, metadata in zip(generated, scores, all_metadata):
            variant = {**score, **metadata, "style": style, "appeal_text": text, "model": used_model,
                       "generator_model": (budget.last_model or used_model) if fallback else used_model,
                       "temperature": 0.5, "generation_ms": generation_ms,
                       "generation_ms_shared": bool(remaining), "scoring_ms": scoring_ms,
                       "generator_status": "fallback" if fallback else "live", "generator_error": error,
                       "gemini_calls_used": state["budget"].calls + state["remaining_budget"].calls,
                       "partial": bool(error or budget.remaining_seconds() <= 0 or (remaining and len(generated) < 2))}
            logger.info("Appeal variant style=%s word_count=%d length_warning=%s", style, len(text.split()), bool(metadata["length_warning"]))
            variants.append(variant)
        logger.info("Appeal stages generation_ms=%d scoring_ms=%.1f suggestions_ms=%.1f repair_ms=0 calls_used=%d",
                    generation_ms, scoring_ms, max((v["suggestions_ms"] for v in variants), default=0), budget.calls)
        variants = guide.ordered_variants(variants, payload.get("length_category", "short"))
        state["results"][stage] = variants
        state["spent_seconds"] = spent + time.monotonic() - started
        return variants


async def repair(text, payload, client="local"):
    language = payload.get("language", "English")
    category = payload.get("length_category", "short")
    started = time.monotonic()
    before = (await asyncio.to_thread(guide.batch_score, [text], language))[0]
    original_scoring_ms = round((time.monotonic() - started) * 1000, 1)
    reservation_day = budgets.improvement_day()
    allowed, remaining = budgets.improvement_uses(client, consume=True, day=reservation_day)
    rejection_reason = None
    after, kept, attempted, message, candidate = before, False, False, None, text
    used_model = None
    candidate_scoring_ms = 0.0
    generation_ms = 0.0
    minimum, maximum = guide.length_band(category)
    if allowed:
        prompt = (f"Rewrite this donation appeal in {guide.quality._normalise_language(language)}. "
                  + guide.length_prompt(category) + " Use ONLY information already in the original appeal. "
                  "Do not add people, claims, deadlines, urgency, amounts, organizations or contacts. Preserve quantities and their units. "
                  "Do not double the length. " + REWRITE_CONSTRAINTS + "Return only the rewrite.\nORIGINAL:\n" + text)
        budget = budgets.Budget(limit=1)
        # Improve is exactly one provider attempt, without fallback/retry.
        attempted = True
        generation_started = time.monotonic()
        try:
            rewrite, used_model = await _call(prompt, budget)
            generation_ms = round((time.monotonic() - generation_started) * 1000, 1)
            facts_ok = is_safe_rewrite(text, rewrite, fallback=_safe_rewrite)
            length_ok = guide.length_warning(rewrite, category) is None
            if facts_ok and length_ok:
                score_started = time.monotonic()
                proposed = (await asyncio.to_thread(guide.batch_score, [rewrite], language))[0]
                candidate_scoring_ms = round((time.monotonic() - score_started) * 1000, 1)
                # Use the same agreement-first ranking for original and Improve candidates.
                # Stable ties keep the original; scoring and guard rules stay unchanged.
                kept = is_visible_improvement(before, proposed)
                if kept:
                    after, candidate = proposed, rewrite
                else:
                    rejection_reason = "no_improvement"
                    message = "Original kept: rewrite did not improve the agreement-first candidate ranking."
            else:
                rejection_reason = "facts_guard" if not facts_ok else "length_guard"
                message = "Original kept: no safe improvement from the information in the appeal."
            if not kept:
                _log_rejected_rewrite(rewrite, rejection_reason,
                                      [name for name, passed in (("facts_guard", facts_ok), ("length_guard", length_ok)) if not passed])
        except (budgets.BudgetExceeded, asyncio.TimeoutError):
            generation_ms = round((time.monotonic() - generation_started) * 1000, 1)
            rejection_reason = "provider_error"
            message = "Original kept: provider unavailable or time/call budget exhausted."
        finally:
            # Provider billing remains recorded; only an accepted rewrite spends the daily allowance.
            if not kept:
                remaining = budgets.refund_improvement(client, reservation_day)
        attempted = budget.attempts > 0
    else:
        message = "Daily improve limit reached. Scoring and suggestions remain available."
    metadata = await asyncio.to_thread(guide.guidance, candidate, language, after, category if "length_category" in payload else None)
    elapsed = round((time.monotonic() - started) * 1000, 1)
    used = budget.calls if allowed else 0
    logger.info("Appeal stage=repair elapsed_ms=%.1f calls_used=%d kept=%s", elapsed, used, kept)
    return {"attempted": attempted, "kept": kept, "before": {"appeal_text": text, **before},
            "after": {"appeal_text": candidate, **after}, "original_appeal": text,
            "original_score": before["quality_score"], "original_raw_quality_score": before["raw_quality_score"],
            "original_label": before["final_label"], "original_confidence": before["confidence"],
            "improved_appeal": candidate, "improved_score": after["quality_score"],
            "improved_raw_quality_score": after["raw_quality_score"], "improved_label": after["final_label"],
            "improved_confidence": after["confidence"], "issues": [], "method": "gated_repair",
            "remaining_improve_uses": remaining, "message": message, "rejection_reason": rejection_reason, "repair_ms": elapsed,
            "generator_model": used_model or (budget.last_model if allowed else None),
            "generator_status": "live" if used_model else "unavailable" if attempted else "not_attempted",
            "provider_attempts": budget.attempts if allowed else 0,
            "candidate_scoring_ms": candidate_scoring_ms,
            "generation_ms": generation_ms, "original_scoring_ms": original_scoring_ms,
            "scoring_ms": round(original_scoring_ms + candidate_scoring_ms, 1),
            "gemini_calls_used": used, **metadata}

"""Local, conservative admission of grammatical rewrites; no models or network.

This supports inflections, clause splitting/deletion and grounded donation requests,
not arbitrary synonyms. Factual equivalence cannot be proved by lexical checks.
Unknown claims fail closed; existing non-Latin handling remains with the caller.
"""
from __future__ import annotations

from dataclasses import dataclass
import logging
import re
import unicodedata
from collections.abc import Callable

logger = logging.getLogger(__name__)
logger.setLevel(logging.INFO)
_WORD = re.compile(r"[^\W\d_]+|\d+(?:[.,]\d+)*", re.UNICODE)
_GRAMMAR = frozenset("a an the and or but for to of in on at by with from into through as such whether that this these those is are was were be been being it its please".split())
_SCOPE = frozenset("not never no without if when unless can could may might must will would should only approximately estimated potentially allegedly".split())
_NEGATION = frozenset("not never no without".split())
_CONTACT = re.compile(r"https?://[^\s<>]+|[\w.+-]+@[\w.-]+\.[a-z]{2,}|\+\d[\d ()-]{6,}\d", re.I)
_MONEY = re.compile(r"(?:[$€£₹]|\b(?:USD|LKR|Rs\.?|EUR|GBP))\s*\d[\d,.]*|\b\d[\d,.]*\s*(?:dollars?|rupees?|USD|LKR|EUR|GBP)\b", re.I)
_DATE = re.compile(r"\b(?:\d{1,4}[-/]\d{1,2}[-/]\d{1,4}|(?:January|February|March|April|May|June|July|August|September|October|November|December)\s+\d{1,2}(?:,?\s+\d{4})?)\b", re.I)
_QUALIFIED_QUANTITY = re.compile(r"\b(?:up to|at most|at least|less than|more than|about|approximately|around|under|over)\s+\d+(?:[.,]\d+)*", re.I)
_TIME_WORDS = frozenset("today tomorrow yesterday tonight immediately urgently urgent now daily weekly monthly annually".split())
_NAME = re.compile(r"\b(?:in|at|from|by|to)\s+([A-Z][a-z]+(?:\s+[A-Z][a-z]+){0,4})|\b((?:[A-Z][\w'-]+\s+){1,4}(?:Foundation|Charity|Trust|Hospital|School))\b")
_IRREGULAR = {"children": "child", "people": "person", "men": "man", "women": "woman"}
_CLAUSE_BREAK = re.compile(r"[.!?;,\u2014\u2013]+(?:\s|$)?|\b(?:but|yet)\b", re.I)
_REQUEST = re.compile(r"^\s*(?:please\s+)?(?:consider\s+)?(?:donat(?:e|ing)|support(?:ing)?|giv(?:e|ing)|contribut(?:e|ing))\b", re.I)
_PURPOSE = re.compile(r"^(.+?)\s+so(?:\s+that)?\s+they\s+(?:can|could|may)\s+(.+)$", re.I)
_STATED_GOAL = re.compile(r"\b(?:dreams?|hopes?|aims?|goals?)\s+(?:of|for|to|is(?:\s+to)?)\s+(.+?)(?=\s+(?:(?:simply\s+)?because|but|although|however|yet)\b|\s+and\s+(?:need|can|cannot|require|lack)\b|[.!?;]|$)", re.I)
_SUBJECT_END = re.compile(r"\b(?:from|in|at|with|who|whose|are|is|were|was|need|needs|can|cannot|could|may|will|would|dream|dreams|hope|hopes)\b", re.I)

REWRITE_CONSTRAINTS = (
    "Improve clarity, remove filler, and end with a clear donation request grounded in the original appeal. "
    "Keep the beneficiaries and donation purpose unchanged. Preserve negated meanings and conditional certainty: "
    "'can provide' must not become 'provides' or 'will provide'. You may split sentences or omit a complete statement, "
    "but never reverse its meaning. Do not add 'today', 'now', a deadline, an amount, or urgency absent from the original. "
    "Reuse the original content words; do not substitute synonyms. Allowed edits are grammatical inflections, "
    "removing filler, splitting sentences, and a grounded final donation request. "
    "A request such as 'Please support these students' is allowed only when those beneficiaries are already stated. "
)


@dataclass(frozen=True)
class RewriteValidation:
    accepted: bool
    reason: str | None = None


def _normalise(text: str) -> str:
    text = unicodedata.normalize("NFKC", text).replace("’", "'")
    text = re.sub(r"\bcan't\b", "cannot", text, flags=re.I)
    text = re.sub(r"\bwon't\b", "will not", text, flags=re.I)
    text = re.sub(r"n't\b", " not", text, flags=re.I)
    return re.sub(r"\bcannot\b", "can not", text, flags=re.I)


def _forms(word: str) -> set[str]:
    """Bounded inflections, without stemming unrelated derivational words."""
    word = word.casefold()
    result = {word, _IRREGULAR.get(word, word)}
    if len(word) > 4 and word.endswith("ies"):
        result.add(word[:-3] + "y")
    if len(word) > 3 and word.endswith("s") and not word.endswith(("ss", "us", "is")):
        result.add(word[:-1])
    if len(word) > 5 and word.endswith("es"):
        result.add(word[:-2])
    for suffix in ("ing",):
        if word.endswith(suffix) and len(word) - len(suffix) >= 3:
            base = word[:-len(suffix)]
            result.update((base, base + "e"))
            if len(base) > 3 and base[-1] == base[-2]:
                result.add(base[:-1])
    return result


def _tokens(text: str) -> list[str]:
    return [w.casefold() for w in _WORD.findall(text) if w.casefold() not in _GRAMMAR]


def _subsequence(candidate: list[str], source: list[str]) -> bool:
    position = 0
    for word in candidate:
        while position < len(source) and not (_forms(word) & _forms(source[position])):
            position += 1
        if position == len(source):
            return False
        position += 1
    return True


def _matches(pattern: re.Pattern, text: str) -> set[str]:
    # URL paths and email local parts can be case-sensitive: preserve contacts.
    return {re.sub(r"\s+", "", m.group() if pattern is _CONTACT else m.group().casefold()).rstrip(".,;!?") for m in pattern.finditer(text)}


def _clauses(text: str) -> list[tuple[str, set[str]]]:
    result = []
    for sentence in re.split(r"[.!?;]+(?:\s|$)", text):
        conditional = re.match(r"\s*(if|when|unless)\b", sentence, re.I)
        inherited = {conditional.group(1).casefold()} if conditional else set()
        result.extend((part.strip(), inherited) for part in _CLAUSE_BREAK.split(sentence) if part.strip())
    return result


def _grounded_purpose_request(request_body: str, original: str) -> bool:
    """Admit an intended benefit, never a promise or a new factual assertion.

    The recipient and benefit must be linked in the same original sentence.
    An inability such as 'cannot afford tuition' can support an appeal to help
    afford tuition; it cannot support an assertion that tuition is now affordable.
    'Pursue' is admitted only with an explicitly stated aspiration's object.
    """
    match = _PURPOSE.fullmatch(request_body.strip())
    if not match:
        return False
    recipient, benefit = _tokens(match[1]), match[2].strip().rstrip(".")
    if not recipient or set(recipient) & _SCOPE:
        return False
    parts = re.split(r"\s+and\s+(?=pursue\b)", benefit, flags=re.I)
    for sentence in re.split(r"[.!?;]+(?:\s|$)", original):
        # Do not borrow an object or aspiration across a contrast introducing
        # another beneficiary or an explicitly unwanted donation purpose.
        sentence = re.split(r"\b(?:but|whereas|while)\b", sentence, maxsplit=1, flags=re.I)[0]
        source = _tokens(sentence)
        subject = _tokens(_SUBJECT_END.split(sentence, maxsplit=1)[0])
        if not _subsequence(recipient, subject):
            continue
        valid = True
        for part in parts:
            words = _tokens(part)
            if not words or set(words) & _SCOPE:
                valid = False
                break
            goal = re.fullmatch(r"pursue\s+(?:their\s+)?(.+)", part.strip(), re.I)
            if goal:
                target = _tokens(goal[1])
                valid = bool(target) and any(
                    not set(w.casefold() for w in _WORD.findall(sentence[:m.start()])[-3:]) & _NEGATION
                    and not set(_tokens(m[1])) & _NEGATION
                    and _subsequence(target, _tokens(m[1]))
                    for m in _STATED_GOAL.finditer(sentence))
            elif set(source) & _NEGATION:
                # Only financial inability is inverted within this intended-benefit
                # request. Other negative facts keep their original meaning.
                inability = re.search(r"\bcan\s+not\s+afford\s+(.+?)(?=[,.!?;]|$)", sentence, re.I)
                valid = len(words) > 1 and words[0] == "afford" and inability is not None and _subsequence(words[1:], _tokens(inability[1]))
            else:
                valid = _subsequence(words, source)
            if not valid:
                break
        if valid:
            return True
    return False


def validate_rewrite(original: str, rewrite: str) -> RewriteValidation:
    """Return a fixed diagnostic code, never source text, keys, or provider data."""
    if not isinstance(original, str) or not isinstance(rewrite, str) or not rewrite.strip():
        return RewriteValidation(False, "empty_rewrite")
    original, rewrite = _normalise(original), _normalise(rewrite)
    if any(c.isalpha() and ord(c) > 127 for c in original + rewrite):
        return RewriteValidation(False, "unsupported_language")
    for pattern, reason in ((_CONTACT, "changed_contact"), (_MONEY, "changed_amount"), (_DATE, "changed_date")):
        if _matches(pattern, original) != _matches(pattern, rewrite):
            return RewriteValidation(False, reason)
    numbers = lambda text: set(re.findall(r"(?<!\w)[+-]?\d+(?:[.,]\d+)*(?:\s*%)?", text))
    if numbers(original) != numbers(rewrite):
        return RewriteValidation(False, "changed_quantity")
    if _matches(_QUALIFIED_QUANTITY, original) != _matches(_QUALIFIED_QUANTITY, rewrite):
        return RewriteValidation(False, "changed_quantity_qualifier")
    names = lambda text: {m.group(1) or m.group(2) for m in _NAME.finditer(text)}
    if names(original) != names(rewrite):
        return RewriteValidation(False, "changed_name")
    before, after = _tokens(original), _tokens(rewrite)
    if not (set(after) & _TIME_WORDS) <= (set(before) & _TIME_WORDS):
        return RewriteValidation(False, "changed_deadline_or_urgency")
    if not (set(after) & _NEGATION) <= (set(before) & _NEGATION):
        return RewriteValidation(False, "changed_negation")
    # Local clauses allow safe sentence splitting. Adjacent source windows retain
    # the subject of relative clauses; never pool vocabulary from the whole appeal.
    clauses = _clauses(original)
    sources = [(" ".join(part for part, _ in clauses[start:start + size]),
                set().union(*(context for _, context in clauses[start:start + size]))) for start in range(len(clauses))
               for size in range(1, min(3, len(clauses) - start) + 1)]
    sentences = [(_tokens(s), bool(re.search(r"\b(?:was|were|had|did)\b", s, re.I)), context) for s, context in sources]
    for sentence, context in _clauses(rewrite):
        content = _tokens(sentence)
        if not content:
            continue
        request = _REQUEST.match(sentence)
        if request:
            # Only an imperative, with an object grounded in the source. New
            # actions, purposes, recipients, amounts and dates remain forbidden.
            target = _tokens(sentence[request.end():])
            if target and not set(target) & _SCOPE and any(_subsequence(target, source) for source, _, _ in sentences):
                continue
            if _grounded_purpose_request(sentence[request.end():], original):
                continue
        past = bool(re.search(r"\b(?:was|were|had|did)\b", sentence, re.I))
        if not any(past == source_past and ((set(content) & _SCOPE) | context) == ((set(source) & _SCOPE) | source_context)
                   and _subsequence(content, source) for source, source_past, source_context in sentences):
            if any(past == source_past and _subsequence(content, source)
                   and set(content) & _NEGATION != set(source) & _NEGATION for source, source_past, _ in sentences):
                return RewriteValidation(False, "changed_negation")
            return RewriteValidation(False, "unsupported_claim_or_changed_relationship")
    return RewriteValidation(True)


def is_safe_rewrite(original: str, rewrite: str, *, fallback: Callable[[str, str], bool]) -> bool:
    result = validate_rewrite(original, rewrite)
    if result.reason == "unsupported_language":
        # No English inflection rules are imposed on Sinhala/Tamil or mixed text.
        return fallback(original, rewrite)
    if not result.accepted:
        logger.info("Appeal rewrite facts_guard detail=%s", result.reason)
    return result.accepted

"""Shared local scoring, confidence metadata and optional evidence-based guidance."""
from __future__ import annotations

import json
import logging
import re

import numpy as np

from app.models.config import settings
from app.models import quality_service as quality

logger = logging.getLogger(__name__)
GAP_PATH = Path(__file__).parent / "data/appeal_gap_statistics.json"
LEVELS = ["low", "borderline", "moderate", "high"]
RANK = {"low": 0, "medium": 1, "high": 2}


def load_gap_statistics():
    if not GAP_PATH.exists():
        return []
    try:
        data = json.loads(GAP_PATH.read_text(encoding="utf-8"))
        if data.get("schema_version") != 1:
            raise ValueError("Unsupported gap statistics schema")
        manifest = json.loads(settings.donation_appeal_model_path("manifest.json").read_text(encoding="utf-8"))
        provenance = data["provenance"]
        if provenance["dataset_sha256"] != manifest["data_sha256"] or provenance["development_rows"] != manifest["development_rows"]:
            raise ValueError("Gap evidence does not match the selected model's development corpus")
        checks = data["checks"]
        if not isinstance(checks, list):
            raise ValueError("Invalid checks")
        for check in checks:
            if not isinstance(check["check"], str) or not isinstance(check["rule"], dict):
                raise ValueError("Invalid rule")
            rule = check["rule"]
            patterns = [rule.get("common_pattern"), *rule.get("patterns", {}).values()]
            patterns.extend(p for values in rule.get("all_patterns", {}).values() for p in values)
            for pattern in patterns:
                if pattern:
                    re.compile(pattern)
            for language in check["enabled_languages"]:
                stat = check["per_language"][language]
                if not stat["selected"] or stat["high_minus_low"] < 0.15:
                    raise ValueError("Unselected gap rule")
                if language == "English" and not all(stat["views"][view]["selected"] for view in ("all_development", "human_written")):
                    raise ValueError("English evidence must pass both robustness views")
        return checks
    except (ValueError, KeyError, OSError, TypeError, re.error):
        logger.warning("Gap statistics unavailable or invalid; using length guidance")
        return []


GAP_CHECKS = load_gap_statistics()


# Shared with the Generator dropdown; ship this component config with the backend.
LENGTH_CONFIG = json.loads((Path(__file__).resolve().parents[3] /
    "frontend/src/config/appealLengths.json").read_text(encoding="utf-8"))


def length_config(category="short"):
    name = str(category).lower().strip().replace(" ", "_")
    return next((value for key, value in LENGTH_CONFIG.items() if name.startswith(key)), LENGTH_CONFIG["short"])


def length_band(category="short"):
    band = length_config(category)
    return band["minimum"], band["maximum"]


def length_prompt(category="short"):
    band = length_config(category)
    return (f"Target {band['target_minimum']}-{band['target_maximum']} words. "
            f"Write at least {band['minimum']} words; maximum {band['maximum']}. "

def confidence_metadata(result):
    probabilities = result.get("class_probs", result.get("probabilities", {})) or {}
    top = sorted(({"class": k, "probability": float(v)} for k, v in probabilities.items()),
                 key=lambda item: (-item["probability"], item["class"]))
    classifier = top[0]["class"] if top else result.get("predicted_class")
    confidence = top[0]["probability"] if top else float(result.get("confidence", 0))
    final = result.get("final_label", result.get("quality_label", result.get("label", result.get("status"))))
    agreement = classifier is not None and final == classifier
    level = (3 if confidence >= settings.CONFIDENCE_HIGH else
             2 if confidence >= settings.CONFIDENCE_MODERATE else
             1 if confidence >= settings.CONFIDENCE_BORDERLINE else 0)
    if not agreement:
        level = max(0, level - 1)
    return {"classifier_top_class": classifier, "classifier_confidence": confidence,
            "label_agreement": agreement, "confidence_level": LEVELS[level], "top_classes": top[:2],
            "agreement_note": "Rating and classifier disagree" if classifier is not None and not agreement else None}


def batch_score(texts, language="English"):
    """One classifier/regressor batch; same artifacts and formulas as quality_service."""
    if not texts:
        return []
    a = quality._load_artifacts()
    language = quality._normalise_language(language)
    texts = [quality._validate_input(text) for text in texts]
    n = len(texts)
    hand = a["scaler"].transform(np.vstack([quality._handcrafted(t, language, a) for t in texts]))
    tf, lsa = [], []
    for key, name in (("en", "English"), ("si", "Sinhala"), ("ta", "Tamil")):
        if language == name:
            features = a["tfidf_" + key].transform(texts)
            tf.append(features)
            lsa.append(a["svd_" + key].transform(features))
        else:
            tf.append(csr_matrix((n, a["N_" + key.upper()])))
            lsa.append(np.zeros((n, a["cfg"]["lsa_components"][key])))
    proba = a["clf"].predict_proba(hstack([csr_matrix(hand), *tf]))
    raw = np.clip(a["reg"].predict(np.hstack([proba, hand, *lsa])), 1, 5)
    scores = np.clip(a["iso"].predict(raw), 1, 5)
    results = []
    for score, probabilities in zip(scores, proba):
        label = str(quality._resolve_label(score, probabilities, a))
        conf = round(float(probabilities.max()), 3)
        result = {"quality_label": label, "final_label": label,
                  "quality_score": quality._clamp_score_to_label(score, label),
                  "raw_quality_score": round(float(score), 2), "confidence": conf,
                  "confidence_normalised": conf, "confidence_display": f"{round(conf * 100)}%",
                  "predicted_class": str(a["le"].classes_[int(np.argmax(probabilities))]),
                  "class_probs": {str(c): round(float(p), 3) for c, p in zip(a["le"].classes_, probabilities)}}
        result.update(confidence_metadata(result))
        results.append(result)
    return results


def length_warning(text, category):
    minimum, maximum = length_band(category)
    words = len(text.split())
    return (f"Generated {words} words; requested band is {minimum}–{maximum} words."
            if words < minimum or words > maximum else None)


def _rule_matches(text, language, rule):
    flags = re.I | re.S
    if "word_count" in rule:
        return rule["word_count"]["min"] <= len(text.split()) and (rule["word_count"]["max"] is None or len(text.split()) <= rule["word_count"]["max"])
    if "all_patterns" in rule:
        return all(re.search(p, text, flags) for p in rule["all_patterns"].get(language, []))
    return any(re.search(p, text, flags) for p in
               [rule.get("common_pattern"), rule.get("patterns", {}).get(language)] if p)


SUGGESTION_MESSAGES = {
    "fund_use": "Say how the donation will be used.",
    "human_detail": "Mention who is affected.",
    "specific_call_to_action": "End with a specific action.",
    "amount_mentioned": "Mention an amount.",
}


def dataset_evidence(check, language="English"):
    stats = check["per_language"][language]
    values = stats.get("views", {}).get("all_development", stats)
    shares = values["classes"]
    return (f"Appears in {shares['high']['share']:.1%} of high-rated "
            f"and {shares['low']['share']:.1%} of low-rated English appeals.")


def guidance(text, language, result, category="short"):
    """Direct text checks only; suggestions never run model inference."""
    started = time.monotonic()
    language = quality._normalise_language(language)
    metadata = confidence_metadata(result)
    final_label = result.get("final_label", result.get("quality_label", result.get("status", result.get("label"))))
    eligible = final_label != "high" or metadata["confidence_level"] == "low" or bool(metadata["agreement_note"])
    minimum = 60 if category is None else length_band(category)[0]
    word_count = len(text.split())
    available = sorted((check for check in GAP_CHECKS
                        if language == "English" and check["check"] in SUGGESTION_MESSAGES
                        and "English" in check.get("enabled_languages", [])),
                       key=lambda check: check["per_language"]["English"]["high_minus_low"], reverse=True)
    suggestions = []
    if eligible:
        if word_count < minimum:
            suggestions.append({"check": "length", "message":
                                (f"This appeal has {word_count} words. High-rated appeals in the data are usually about 60 or more."
                                 if category is None else f"This appeal has {word_count} words. Appeals under about {minimum} words tend to rate lower."),
                                "evidence": ""})
        gaps = [check for check in available if not _rule_matches(text, language, check["rule"])][:2]
        suggestions.extend({"check": check["check"], "message": SUGGESTION_MESSAGES[check["check"]],
                            "evidence": dataset_evidence(check)} for check in gaps)
    elapsed = round((time.monotonic() - started) * 1000, 1)
    logger.info("Appeal stage=suggestions elapsed_ms=%.1f checks=%d", elapsed, len(suggestions))
    return {**metadata, "suggestions": suggestions[:3], "gap_statistics_available": bool(available),
            "suggestions_ms": elapsed, "word_count": word_count, "length_warning": length_warning(text, category) if category is not None else None,
            "dataset_gap_status": "not surfaced: likely augmentation artifact" if language != "English" else "available" if available else "unavailable"}


def guidance_many(texts, language, scores, category="short"):
    return [guidance(text, language, score, category) for text, score in zip(texts, scores)]


def ordered_variants(variants, category="short"):
    """Selected word band, agreement, label, classifier confidence, then score."""
    def key(variant):
        metadata = confidence_metadata(variant)
        minimum, maximum = length_band(category)
        words = variant.get("word_count", len(variant.get("appeal_text", "").split()))
        return (minimum <= words <= maximum, variant.get("label_agreement", metadata["label_agreement"]) is True,
                RANK.get(variant["final_label"], -1),
                float(variant.get("classifier_confidence", metadata["classifier_confidence"])),
                float(variant["quality_score"]))
    return sorted(variants, key=key, reverse=True)


def recommended(variants, category="short"):
    return next(iter(ordered_variants(variants, category)), None)

"""
Donation appeal quality scoring service.

Pipeline:
text + language -> handcrafted features + per-language TF-IDF/LSA
-> calibrated classifier probabilities -> GBR regressor -> isotonic score.
"""

from __future__ import annotations

import asyncio
import logging
import re

import joblib
import numpy as np
from scipy.sparse import csr_matrix, hstack

from app.models.config import settings
from app.services.mongo_service import log_appeal_analysis


logger = logging.getLogger(__name__)

_ARTIFACTS: dict = {}


class QualityModelLoadError(RuntimeError):
    """The selected appeal model cannot be loaded."""

_LOW_CONFIDENCE_THRESHOLD = 0.50
_MIN_TEXT_LENGTH = 10
_MAX_TEXT_LENGTH = 10_000

_LANGUAGE_ALIASES = {
    "en": "English",
    "eng": "English",
    "english": "English",
    "si": "Sinhala",
    "sin": "Sinhala",
    "sinhala": "Sinhala",
    "ta": "Tamil",
    "tam": "Tamil",
    "tamil": "Tamil",
}


def _normalise_language(language: str | None) -> str:
    value = (language or "English").strip().lower()
    return _LANGUAGE_ALIASES.get(value, "English")


def _load_artifacts() -> dict:
    global _ARTIFACTS
    if _ARTIFACTS:
        return _ARTIFACTS

    names = {
        "tfidf_en": "appeal_quality_vectorizer.joblib",
        "tfidf_si": "appeal_quality_vectorizer_si.joblib",
        "tfidf_ta": "appeal_quality_vectorizer_ta.joblib",
        "svd_en": "appeal_quality_svd_en.joblib",
        "svd_si": "appeal_quality_svd_si.joblib",
        "svd_ta": "appeal_quality_svd_ta.joblib",
        "scaler": "appeal_quality_scaler.joblib",
        "clf": "appeal_quality_classifier.joblib",
        "le": "appeal_quality_label_encoder.joblib",
        "reg": "appeal_quality_regressor.joblib",
        "iso": "appeal_quality_isotonic.joblib",
        "cfg": "appeal_quality_config.joblib",
    }

    model_dir = settings.donation_appeal_model_path("")
    if not model_dir.is_dir():
        raise QualityModelLoadError(
            f"Appeal quality model directory missing: {model_dir}. "
            "Restore the selected model or correct APPEAL_MODEL_DIR; "
            "no alternate model will be loaded."
        )
    missing = [
        filename for filename in names.values()
        if not (model_dir / filename).is_file()
    ]
    if missing:
        raise QualityModelLoadError(
            f"Appeal quality model is incomplete at {model_dir}: "
            + ", ".join(missing)
        )

    loaded = {}
    for key, filename in names.items():
        path = model_dir / filename
        try:
            loaded[key] = joblib.load(path)
        except Exception as exc:
            raise QualityModelLoadError(
                f"Cannot load appeal quality artifact: {path} "
                f"({type(exc).__name__})"
            ) from exc
        logger.info("Loaded donation appeal artifact: %s", filename)

    loaded["N_EN"] = loaded["tfidf_en"].transform([""]).shape[1]
    loaded["N_SI"] = loaded["tfidf_si"].transform([""]).shape[1]
    loaded["N_TA"] = loaded["tfidf_ta"].transform([""]).shape[1]
    loaded["ZERO_EN"] = csr_matrix((1, loaded["N_EN"]), dtype=np.float64)
    loaded["ZERO_SI"] = csr_matrix((1, loaded["N_SI"]), dtype=np.float64)
    loaded["ZERO_TA"] = csr_matrix((1, loaded["N_TA"]), dtype=np.float64)

    _ARTIFACTS = loaded
    cfg = loaded["cfg"]
    logger.info(
        "Donation appeal quality scoring pipeline ready (version=%s, trained_at=%s)",
        cfg.get("model_version", "unknown"),
        cfg.get("trained_at", "unknown"),
    )
    return _ARTIFACTS


def warm_up() -> None:
    """Eagerly load all quality-scoring artifacts at application startup."""
    _load_artifacts()
    logger.info("Donation appeal quality scoring service warmed up.")


def _validate_input(text: str) -> str:
    """Validate and sanitise appeal text before running ML inference."""
    if not text or not text.strip():
        raise ValueError("Appeal text cannot be empty.")

    stripped = text.strip()
    if len(stripped) < _MIN_TEXT_LENGTH:
        raise ValueError(
            f"Appeal text is too short to score reliably "
            f"(minimum {_MIN_TEXT_LENGTH} characters, got {len(stripped)})."
        )

    if len(stripped) > _MAX_TEXT_LENGTH:
        logger.warning(
            "Appeal text exceeds maximum length (%d chars); truncating to %d.",
            len(stripped),
            _MAX_TEXT_LENGTH,
        )
        stripped = stripped[:_MAX_TEXT_LENGTH]

    return stripped


def _handcrafted(text: str, language: str, artifacts: dict) -> np.ndarray:
    t = str(text or "")
    words = t.split()
    sentence_count = len(re.findall(r"[.!?।]", t))
    avg_sent_denominator = len(re.findall(r"[.!?།]", t)) + 2

    row = {
        "log_text_len": np.log1p(len(t)),
        "log_word_count": np.log1p(len(words)),
        "avg_word_len": len(t) / (len(words) + 1),
        "sentence_count": np.log1p(sentence_count + 1),
        "avg_sent_len": len(words) / avg_sent_denominator,
        "exclamation_count": np.log1p(t.count("!")),
        "question_count": np.log1p(t.count("?")),
        "digit_ratio": len(re.findall(r"\d", t)) / (len(t) + 1),
        "upper_ratio": len(re.findall(r"[A-Z]", t)) / (len(t) + 1),
        "has_currency": int(bool(re.search(r"[$£€Rs\d]", t))),
        "has_donate_word": int(
            bool(
                re.search(
                    r"donat|contribut|help|support|give|fund|රුපියල්|දීමනා|நன்கொடை",
                    t,
                    re.IGNORECASE,
                )
            )
        ),
        "is_sinhala": int(language == "Sinhala"),
        "is_tamil": int(language == "Tamil"),
    }

    feature_names = artifacts["cfg"]["scaler_feature_names"]
    return np.array([row[name] for name in feature_names], dtype=float).reshape(1, -1)


def _resolve_label(score: float, proba: np.ndarray, artifacts: dict) -> str:
    tolerance = artifacts["cfg"].get("boundary_tolerance", 0.25)

    if score <= 2.0:
        score_label = "low"
    elif score <= 3.0:
        score_label = "medium"
    else:
        score_label = "high"

    clf_label = artifacts["le"].classes_[int(np.argmax(proba))]
    if score_label == clf_label:
        return score_label

    if abs(score - 2.0) < tolerance or abs(score - 3.0) < tolerance:
        return clf_label

    return score_label


def _rule_based_score(text: str, language: str) -> dict:
    """
    Rule-based fallback scorer using handcrafted signals only.
    Runs when the ML pipeline is unavailable, so no artifacts are required.
    """
    t = str(text or "").strip()
    words = t.split()
    word_count = len(words)

    score = 1.0

    if word_count >= 20:
        score += 0.5
    if word_count >= 60:
        score += 0.5
    if word_count >= 120:
        score += 0.3
    if word_count > 300:
        score -= 0.3

    if re.search(r"[$£€Rs]\s?\d|Rs\.?\s?\d|\d+,\d{3}", t):
        score += 0.5

    if re.search(
        r"donat|contribut|help|support|give|fund|රුපියල්|දීමනා|நன்கொடை",
        t,
        re.IGNORECASE,
    ):
        score += 0.4

    if re.search(
        r"urgent|immediately|now|today|critical|අවශ්‍ය|இப்போது",
        t,
        re.IGNORECASE,
    ):
        score += 0.3

    exclamations = t.count("!")
    if 1 <= exclamations <= 3:
        score += 0.2

    score = float(round(min(max(score, 1.0), 5.0), 2))

    if score <= 2.0:
        label = "low"
    elif score <= 3.0:
        label = "medium"
    else:
        label = "high"

    logger.warning(
        "ML pipeline unavailable; using rule-based fallback scorer "
        "(language=%s, score=%.2f, label=%s)",
        language,
        score,
        label,
    )

    return {
        "quality_label": label,
        "predicted_class": label,
        "quality_score": score,
        "raw_quality_score": score,
        "confidence": 0.0,
        "low_confidence": True,
        "probabilities": {},
        "method": "rule_based",
    }


def _clamp_score_to_label(score: float, label: str) -> float:
    lower, upper = {
        "low": (1.0, 2.0),
        "medium": (2.01, 3.0),
        "high": (3.01, 5.0),
    }[label]
    return round(float(np.clip(score, lower, upper)), 2)


def score_appeal(text: str, language: str = "English") -> dict:
    text = _validate_input(text)

    try:
        artifacts = _load_artifacts()
        language = _normalise_language(language)
        lsa_components = artifacts["cfg"]["lsa_components"]

        hand_raw = _handcrafted(text, language, artifacts)
        hand_scaled = artifacts["scaler"].transform(hand_raw)

        if language == "Sinhala":
            tfidf_feat = artifacts["tfidf_si"].transform([text])
            lsa_en = np.zeros((1, lsa_components["en"]))
            lsa_si = artifacts["svd_si"].transform(tfidf_feat)
            lsa_ta = np.zeros((1, lsa_components["ta"]))
            clf_tfidf_en = artifacts["ZERO_EN"]
            clf_tfidf_si = tfidf_feat
            clf_tfidf_ta = artifacts["ZERO_TA"]
        elif language == "Tamil":
            tfidf_feat = artifacts["tfidf_ta"].transform([text])
            lsa_en = np.zeros((1, lsa_components["en"]))
            lsa_si = np.zeros((1, lsa_components["si"]))
            lsa_ta = artifacts["svd_ta"].transform(tfidf_feat)
            clf_tfidf_en = artifacts["ZERO_EN"]
            clf_tfidf_si = artifacts["ZERO_SI"]
            clf_tfidf_ta = tfidf_feat
        else:
            tfidf_feat = artifacts["tfidf_en"].transform([text])
            lsa_en = artifacts["svd_en"].transform(tfidf_feat)
            lsa_si = np.zeros((1, lsa_components["si"]))
            lsa_ta = np.zeros((1, lsa_components["ta"]))
            clf_tfidf_en = tfidf_feat
            clf_tfidf_si = artifacts["ZERO_SI"]
            clf_tfidf_ta = artifacts["ZERO_TA"]

        x_clf = hstack(
            [csr_matrix(hand_scaled), clf_tfidf_en, clf_tfidf_si, clf_tfidf_ta]
        )
        proba = artifacts["clf"].predict_proba(x_clf)[0]
        predicted_class = str(artifacts["le"].classes_[int(np.argmax(proba))])

        x_reg = np.hstack([proba.reshape(1, -1), hand_scaled, lsa_en, lsa_si, lsa_ta])
        raw_score = float(np.clip(artifacts["reg"].predict(x_reg)[0], 1.0, 5.0))
        score = float(np.clip(artifacts["iso"].predict([raw_score])[0], 1.0, 5.0))
        label = _resolve_label(score, proba, artifacts)
        raw_quality_score = round(score, 2)
        quality_score = _clamp_score_to_label(score, label)

        confidence = round(float(proba.max()), 3)
        low_confidence = confidence < _LOW_CONFIDENCE_THRESHOLD
        if low_confidence:
            logger.warning(
                "Low-confidence prediction (%.3f) for %s text; consider human review.",
                confidence,
                language,
            )

        return {
            "quality_label": label,
            "predicted_class": predicted_class,
            "quality_score": quality_score,
            "raw_quality_score": raw_quality_score,
            "confidence": confidence,
            "low_confidence": low_confidence,
            "probabilities": {
                cls: round(float(probability), 3)
                for cls, probability in zip(artifacts["le"].classes_, proba)
            },
        }
    except QualityModelLoadError:
        raise
    except Exception as exc:
        logger.error("ML inference failed; activating rule-based fallback: %s", exc)
        language = _normalise_language(language)
        return _rule_based_score(text, language)


def get_quality_score(appeal_text: str, language: str = "English") -> dict:
    """
    Backward-compatible API used by FastAPI and Gemini appeal generation.
    """
    try:
        result = score_appeal(appeal_text, language)
        confidence = result["confidence"]
        response = {
            "score": result["quality_score"],
            "raw_quality_score": result["raw_quality_score"],
            "status": result["quality_label"],
            "predicted_class": result.get("predicted_class", result["quality_label"]),
            "class_probs": result.get("probabilities", {}),
            "method": result.get("method", "ml_model"),
            "confidence": confidence,
            "confidence_normalised": confidence,
            "confidence_display": f"{round(confidence * 100)}%",
            "low_confidence": result["low_confidence"],
            "probabilities": result["probabilities"],
        }

        try:
            loop = asyncio.get_running_loop()
            loop.create_task(
                log_appeal_analysis(appeal_text, language, response, source="api")
            )

def _retry_delay_seconds(exc: Exception, message: str) -> float:
    delays = [0.0]

    def add(value):
        try:
            seconds = float(str(value).removesuffix("s"))
            if math.isfinite(seconds) and seconds >= 0:
                delays.append(seconds)
        except (TypeError, ValueError):
            pass

    response = getattr(exc, "response", None)
    headers = getattr(response, "headers", {}) or {}
    retry_after = headers.get("Retry-After")
    if retry_after:
        try:
            add(float(retry_after))
        except (TypeError, ValueError):
            try:
                when = parsedate_to_datetime(retry_after)
                if when.tzinfo is None:
                    when = when.replace(tzinfo=timezone.utc)
                add((when - datetime.now(timezone.utc)).total_seconds())
            except (TypeError, ValueError, OverflowError):
                pass
    body = getattr(exc, "details", {}) or {}
    error = body.get("error", body) if isinstance(body, dict) else {}
    if isinstance(error, dict):
        for detail in error.get("details", []) or []:
            if isinstance(detail, dict) and str(detail.get("@type", "")).endswith("RetryInfo"):
                add(detail.get("retryDelay"))
    match = re.search(r"\bretry in\s+(\d+(?:\.\d+)?)\s*s", message)
    if match:
        add(match.group(1))
    return max(delays)

def _cache_key(prompt: str, temp: float) -> str:
    return hashlib.sha256(f"{temp:.3f}|{prompt}".encode()).hexdigest()


def _get_cached(prompt: str, temp: float) -> dict | None:
    key = _cache_key(prompt, temp)
    entry = _RESULT_CACHE.get(key)
    if entry is None:
        return None
    result, ts = entry
    if time.time() - ts > _RESULT_CACHE_TTL:
        del _RESULT_CACHE[key]
        return None
    return result


def _store_cache(prompt: str, temp: float, result: dict) -> None:
    _RESULT_CACHE[_cache_key(prompt, temp)] = (result, time.time())

def gemini_health_check() -> dict:
    """
    Lightweight check: verifies google.genai imports and hits the models-list
    endpoint (NOT generateContent). Result is cached for 60 seconds.
    Never logs or returns the API key.
    """
    now = time.time()
    if _GEMINI_HEALTH and now - _GEMINI_HEALTH.get("ts", 0) < 60.0:
        return {k: v for k, v in _GEMINI_HEALTH.items() if k != "ts"}

    # 1. Import check
    try:
        from google import genai as _  # noqa: F401
    except ImportError as exc:
        result = {
            "status": "unavailable",
            "message": f"google-genai not importable in {sys.executable}: {exc}",
        }
        _GEMINI_HEALTH.update({**result, "ts": now})
        return result

    # 2. Key configured?
    if not settings.GEMINI_API_KEY:
        result = {"status": "unconfigured", "message": "GEMINI_API_KEY not set"}
        _GEMINI_HEALTH.update({**result, "ts": now})
        return result

    # 3. Models-list probe (not generateContent — free metadata call)
    http_status: int | None = None
    try:
        url = (
            "https://generativelanguage.googleapis.com/v1beta/models"
            f"?key={settings.GEMINI_API_KEY}"
        )
        req = urllib.request.Request(url)
        with urllib.request.urlopen(req, timeout=5) as resp:
            http_status = resp.status
    except urllib.error.HTTPError as exc:
        http_status = exc.code
    except Exception:
        # Deliberately vague — don't expose URL (contains key)
        result = {"status": "error", "message": "Network error reaching Gemini API"}
        _GEMINI_HEALTH.update({**result, "ts": now})
        return result

    if http_status == 200:
        result = {"status": "reachable", "message": "models-list HTTP 200"}
    elif http_status == 403:
        result = {"status": "auth_error", "message": f"HTTP {http_status} — key rejected"}
    elif http_status == 429:
        result = {"status": "quota_exceeded", "message": f"HTTP {http_status} — rate limited"}
    else:
        result = {"status": "error", "message": f"HTTP {http_status}"}

    _GEMINI_HEALTH.update({**result, "ts": now})
    return result

def _has_hallucination(text: str) -> tuple[bool, str]:
    """Return True with a reason when generated text drifts from factual copy."""
    for pattern in _INVENTED_PERSON_PATTERNS:
        match = pattern.search(text)
        if match:
            return True, f"invented_person: {match.group()!r}"

    for pattern in _INVENTED_SPECIFIC_PATTERNS:
        match = pattern.search(text)
        if match:
            return True, f"invented_specific: {match.group()!r}"

    filler_hits = []
    for pattern in _FILLER_PATTERNS:
        match = pattern.search(text)
        if match:
            filler_hits.append(match.group())

    if len(filler_hits) >= _FILLER_THRESHOLD:
        return True, f"generic_filler: {filler_hits}"

    return False, ""


def _has_language_mixing(text: str, expected_language: str) -> bool:
    """Return True when a non-English appeal contains significant Latin script."""
    if expected_language not in ("Sinhala", "Tamil"):
        return False
    non_ws = re.sub(r"\s", "", text)
    if not non_ws:
        return False
    latin_chars = sum(1 for ch in non_ws if ch.isascii() and ch.isalpha())
    return (latin_chars / len(non_ws)) > 0.15


def _has_invented_person(text: str) -> bool:
    """Return True if the text contains hallucinated individuals or narrative drift."""
    has_problem, _reason = _has_hallucination(text)
    return has_problem


def _detect_script(text: str) -> str:
    """Detect the dominant script from Unicode ranges."""
    sinhala_chars = sum(1 for char in text if "\u0D80" <= char <= "\u0DFF")
    tamil_chars = sum(1 for char in text if "\u0B80" <= char <= "\u0BFF")
    latin_chars = sum(1 for char in text if char.isascii() and char.isalpha())
    total = max(len(text), 1)

    if sinhala_chars / total > 0.1:
        return "Sinhala"
    if tamil_chars / total > 0.1:
        return "Tamil"
    if latin_chars / total > 0.3:
        return "English"
    return "unknown"


def _language_matches(text: str, expected_language: str) -> bool:
    """Return True when generated text matches the requested language."""
    detected = _detect_script(text)
    if expected_language == "English" and detected == "English":
        return True
    if expected_language == "Sinhala" and detected == "Sinhala":
        return True
    if expected_language == "Tamil" and detected == "Tamil":
        return True
    if detected == "unknown":
        return True
    return False


def score_text(text: str, language: str = "English") -> dict:
    """
    Delegate entirely to quality_service: single inference pipeline,
    no duplicate artifact loading.
    """
    result = get_quality_score(text, language)
    return {
        "quality_label": result["status"],
        "final_label": result["status"],
        "predicted_class": result.get("predicted_class"),
        "class_probs": result.get("class_probs", {}),
        "quality_score": result["score"],
        "raw_quality_score": result.get("raw_quality_score", result["score"]),
        "confidence": result["confidence"],
        "confidence_normalised": result["confidence_normalised"],
        "confidence_display": result["confidence_display"],
    }


async def call_gemini_with_key(
    prompt: str, temp: float, api_key: str | None, *, model: str | None = None,
) -> str:
    def _generate() -> str:
        import requests
        from google.genai.errors import APIError
        from app.services.appeal_budget import CURRENT

        budget = CURRENT.get()
        remaining = budget.remaining_seconds() if budget else None
        http_timeout = (min(5.0, max(0.05, remaining / 4)), min(15.0, max(0.05, remaining * 3 / 4))) if budget else GEMINI_HTTP_TIMEOUT

        key = api_key or settings.GEMINI_API_KEY
        if not key:
            return ""
        # The installed SDK cannot serialize thinkingLevel; use the verified REST body.
        body = {
            "contents": [{"parts": [{"text": f"{SYSTEM_INSTRUCTION}\n\n{prompt}"}]}],
            "generationConfig": {
                "temperature": temp,
                "maxOutputTokens": 2048 if budget else 512,
                "thinkingConfig": {"thinkingLevel": "MINIMAL"},
            },
        }
        with requests.Session() as session:
            response = session.request(
                "POST",
                "https://generativelanguage.googleapis.com/v1beta/models/"
                f"{model or settings.GEMINI_MODEL}:generateContent",
                headers={"x-goog-api-key": key, "Content-Type": "application/json"},
                json=body,
                timeout=http_timeout,
            )
            APIError.raise_for_response(response)
            return "".join(
                part.get("text", "")
                for candidate in response.json().get("candidates", [])
                for part in candidate.get("content", {}).get("parts", [])
                if not part.get("thought")
            ).strip()
    return await asyncio.to_thread(_generate)


async def call_gemini(prompt: str, temp: float) -> str:
    return await call_gemini_with_key(prompt, temp, settings.GEMINI_API_KEY)


def _call_hf_llama_sync_with_temp(prompt: str, temperature: float) -> str:
    """
    Llama 3.2 3B via featherless-ai provider on HuggingFace.
    Uses HF_API_KEY; generated text is still validated before being returned.
    """
    from huggingface_hub import InferenceClient

    client = InferenceClient(
        token=settings.HF_API_KEY,
        provider="featherless-ai",
        timeout=HF_HTTP_TIMEOUT,
    )
    response = client.chat_completion(
        model="meta-llama/Llama-3.2-3B-Instruct",
        messages=[
            {
                "role": "system",
                "content": SYSTEM_INSTRUCTION,
            },
            {
                "role": "user",
                "content": prompt,
            },
        ],
        max_tokens=512,
        temperature=temperature,
    )
    return response.choices[0].message.content.strip()


def _call_hf_llama_sync(prompt: str) -> str:
    return _call_hf_llama_sync_with_temp(prompt, 0.3)


async def call_gemini_with_key_fallback(
    prompt: str,
    temp: float,
    *,
    language: str = "English",
    max_retries: int = 2,
    key_offset: int = 0,
) -> dict:
    # Fast-path: return cached result for identical inputs
    cached = _get_cached(prompt, temp)
    if cached:
        logger.debug("Cache hit for generation request (temp=%.2f)", temp)
        return cached

    # Fail fast: if the library is missing, don't burn through all 11 keys
    try:
        from google import genai as _genai_check  # noqa: F401
    except ImportError as exc:
        logger.error(
            "google-genai not installed in %s — start the server from backend/venv. Error: %s",
            sys.executable, exc,
        )
        return {"generator_error": "google-genai unavailable", "generator_model": settings.GEMINI_MODEL}

    # Mock mode: return canned appeal, no network call
    if settings.GENERATOR_MODE == "mock":
        result = {
            "model": "mock",
            "temperature": 0.0,
            "appeal_text": (
                "MOCK: Families in the affected area urgently need support. "
                "This campaign aims to help those displaced. Please donate today."
            ),
            "provider": "mock",
            "generator_status": "live",
            "generator_error": None,
        }
        _store_cache(prompt, temp, result)
        return result

    from app.services.credit_monitor import _key_state

    all_keys = list(enumerate(_gemini_api_keys(), start=1))
    if all_keys:
        with _KEY_ORDER_LOCK:
            start = next(_KEY_STARTS)
        offset = (start + key_offset) % len(all_keys)
        gemini_keys = all_keys[offset:] + all_keys[:offset]
    else:
        gemini_keys = []
    models = list(dict.fromkeys([
        settings.GEMINI_MODEL,
        *(name.strip().removeprefix("models/")
          for name in settings.GEMINI_FALLBACK_MODELS.split(",") if name.strip()),
    ]))
    failure = {"generator_error": "No usable Gemini keys configured", "generator_model": settings.GEMINI_MODEL}
    keys_tried = 0
    for key_index, api_key in gemini_keys:
        label = f"gemini_key_{key_index}"
        if is_exhausted(label):
            continue
        if keys_tried >= settings.GEMINI_MAX_KEYS_PER_REQUEST:
            break
        keys_tried += 1
        rotate_key = False
        for model_index, model in enumerate(models):
            # A primary 503 gets one short retry; other transient codes keep their policy.
            # Each fallback model gets exactly one call on the same key.
            attempts = 4 if model_index == 0 else 1
            retry_temp = temp
            for attempt in range(1, attempts + 1):
                try:
                    if model_index == 0:
                        text = await call_gemini_with_key(prompt, retry_temp, api_key)
                    else:
                        text = await call_gemini_with_key(prompt, retry_temp, api_key, model=model)
                    logger.info(
                        "Gemini attempt model=%s key=%d attempt=%d/%d http_status=200 next_delay=0.000s",
                        model, key_index, attempt, attempts,
                    )
                    text = text.strip()
                    if not text:
                        cause = "Empty Gemini response"
                    elif _has_language_mixing(text, language):
                        cause = "Generated text failed language validation"
                    elif _has_hallucination(text)[0]:
                        cause = "Generated text failed factual validation"
                    else:
                        record_success(label)
                        result = {
                            "model": model, "generator_model": model,
                            "temperature": retry_temp, "appeal_text": text,
                            "provider": label, "generator_status": "live",
                            "generator_error": None,
                        }
                        _store_cache(prompt, temp, result)
                        return result
                    failure = {"generator_error": cause, "generator_model": model}
                    # Preserve the caller's limit for content-validation attempts.
                    if attempt >= max_retries:
                        break
                    retry_temp = max(0.1, retry_temp - 0.2)
                except Exception as exc:
                    if isinstance(exc, (ImportError, ModuleNotFoundError)):
                        logger.info("Gemini failure key=%d http_status=none quota_reason=unspecified", key_index)
                        logger.info(
                            "Gemini attempt model=%s key=%d attempt=%d/%d http_status=none next_delay=0.000s",
                            model, key_index, attempt, attempts,
                        )
                        return {"generator_error": type(exc).__name__, "generator_model": model}
                    message = str(exc).lower()
                    code = getattr(exc, "code", None) or getattr(exc, "status_code", None)
                    if code is None:
                        match = re.search(r"\b(400|401|403|404|429|500|503|504)\b", message)
                        code = int(match.group(1)) if match else None
                    try:
                        code = int(code) if code is not None else None
                    except (TypeError, ValueError):
                        code = None
                    unsupported_thinking = code == 400 and "thinking" in message
                    overloaded = code in (500, 503, 504)
                    daily = code == 429 and bool(_DAILY_QUOTA_RE.search(message))
                    transient = overloaded or (code == 429 and not daily)
                    cause = ("400 unsupported thinking configuration" if unsupported_thinking else
                             f"{code} overloaded" if overloaded else
                             "429 daily quota exhausted" if daily else
                             "429 per-minute rate limit" if code == 429 else
                             "404 model not found" if code == 404 else
                             f"{code} key rejected" if code in (400, 401, 403) else
                             type(exc).__name__)
                    # Never log provider exception text: it can contain credentials.
                    hint = _retry_delay_seconds(exc, message) if transient else 0.0
                    rotate_now = (code == 429 and not daily and not (max_retries >= 3 and hint <= 10)) or (
                        transient and code != 503 and hint > 10.0
                    )
                    if code == 503:
                        # Overload is model-wide: never rotate keys for a 503.
                        # A longer provider hint means move to the next model now.
                        retry_503 = model_index == 0 and attempt == 1 and hint <= PRIMARY_503_RETRY_DELAY
                        delay = PRIMARY_503_RETRY_DELAY if retry_503 else 0.0
                    else:
                        delay = (
                            min(10.0, max(2.0 ** attempt + random.uniform(0, 1), hint))
                            if transient and attempt < attempts and not rotate_now else 0.0
                        )
                    logger.info(
                        "Gemini failure key=%d http_status=%s quota_reason=%s",
                        key_index, code or "none", _quota_reason(message)[:120],
                    )
                    logger.info(
                        "Gemini attempt model=%s key=%d attempt=%d/%d http_status=%s next_delay=%.3fs",
                        model, key_index, attempt, attempts, code or "none", delay,
                    )
                    failure = {
                        "generator_error": cause + (f" after {attempt - 1} retries" if transient else ""),
                        "generator_model": model,
                    }
                    if rotate_now:
                        record_failure(label, cause)
                        rotate_key = True
                        break
                    if code == 503:
                        if retry_503:
                            await asyncio.sleep(delay)
                            continue
                        break  # Next model on the same key, including long retry hints.
                    if transient:
                        if attempt < attempts:
                            await asyncio.sleep(delay)
                            continue
                        break  # Try the next model, never another key for overload.
                    if daily or (code in (400, 401, 403) and not unsupported_thinking):
                        mark_exhausted(label)
                        rotate_key = True
                    elif code != 404 and not unsupported_thinking:
                        record_failure(label, cause)
                        rotate_key = True
                    break
            if rotate_key:
                break
        if not rotate_key:
            # Model-wide failures must not penalize or rotate API keys.
            return failure

    if settings.HF_API_KEY:
        if language in ("Sinhala", "Tamil"):
            logger.warning(
                "HF Llama skipped for %s — cannot reliably generate non-Latin scripts; "
                "using template fallback",
                language,
            )
            return failure

        logger.info("All Gemini keys exhausted — trying HuggingFace Llama fallback")
        _HF_LABEL = "huggingface-featherless"

        for hf_attempt, hf_temp in enumerate((0.3, 0.2, 0.1), start=1):
            try:
                appeal_text = await asyncio.to_thread(
                    _call_hf_llama_sync_with_temp, prompt, hf_temp,
                )
                appeal_text = appeal_text.strip()
                if not appeal_text:
                    continue
                if not _language_matches(appeal_text, language):
                    logger.warning(
                        "HF Llama wrong language (expected=%s, attempt=%d) — rejecting",
                        language, hf_attempt,
                    )
                    continue
                has_problem, reason = _has_hallucination(appeal_text)
                if not has_problem:
                    logger.info("HF Llama succeeded on attempt %d", hf_attempt)
                    record_success(_HF_LABEL)
                    result = {
                        "model": "llama-3.2-3b-instruct",
                        "temperature": hf_temp,
                        "appeal_text": appeal_text,
                        "provider": _HF_LABEL,
                        "generator_status": "live",
                        "generator_error": None,
                    }
                    _store_cache(prompt, temp, result)
                    return result
                logger.warning(
                    "HF Llama hallucination (attempt=%d, temp=%.1f): %s",
                    hf_attempt, hf_temp, reason,
                )
            except Exception as exc:
                error_str = str(exc).lower()
                if any(kw in error_str for kw in _PERMANENT_QUOTA_KEYWORDS):
                    logger.warning("HF Llama quota exhausted: %s", type(exc).__name__)
                    mark_exhausted(_HF_LABEL)
                elif any(kw in error_str for kw in _TEMPORARY_RATE_KEYWORDS):
                    logger.warning("HF Llama rate limited: %s", type(exc).__name__)
                    record_failure(_HF_LABEL, "rate_limited_temporary")
                else:
                    logger.error("HF Llama attempt %d failed: %s", hf_attempt, type(exc).__name__)
                    record_failure(_HF_LABEL, type(exc).__name__)
                break

        logger.warning("HF Llama fallback exhausted or all attempts hallucinated")

    return failure


def _build_fallback_appeal(form_data: dict) -> str:
    fallback_templates = {
        "Sinhala": (
            "{location} ප්‍රදේශයේ පවුල් වලට {need} අවශ්‍ය වේ. "
            "මෙම {campaign_type} ව්‍යාපාරය {goal} අරමුණු කරයි. "
            "අපේ ප්‍රජාව එකට ඉදිරියට යමු. අදම පරිත්‍යාග කරන්න."
        ),
        "Tamil": (
            "{location} பகுதியில் உள்ள குடும்பங்களுக்கு {need} தேவைப்படுகிறது. "
            "இந்த {campaign_type} பிரச்சாரம் {goal} நோக்கமாக கொண்டுள்ளது. "
            "நம் சமூகம் ஒன்றாக முன்னேறுவோம். இன்றே நன்கொடை வழங்குங்கள்."
        ),
        "English": (
            "Families in {location} urgently need {need}. "
            "This {campaign_type} campaign aims to {goal}. "
            "Our community is coming together to help. Please donate today."
        ),
    }
    location      = form_data.get("location") or "our community"
    need          = form_data.get("verified_need") or "urgent support"
    campaign_type = (form_data.get("campaign_type") or "relief").replace("_", " ")
    language      = form_data.get("language") or "English"
    goal          = form_data.get("campaign_goal") or "help those affected recover"
    if need.strip().lower() == "cloths":
        need = "clothes"
    # Lowercase first character so "Support families" → "support families" mid-sentence
    if language in ("en", "English") and goal:
        goal = goal[:1].lower() + goal[1:]

    template = fallback_templates.get(language, fallback_templates["English"])
    return template.format(
        location=location,
        need=need,
        goal=goal,
        campaign_type=campaign_type,
    )

        "generator_model": generator_model,
        "temperature": 0.0,
        "style": "fallback",
        "appeal_text": appeal_text,
        "quality_label": scoring["quality_label"],
        "final_label": scoring["quality_label"],
        "predicted_class": scoring.get("predicted_class"),
        "class_probs": scoring.get("class_probs", {}),
        "quality_score": scoring["quality_score"],
        "raw_quality_score": scoring.get("raw_quality_score", scoring["quality_score"]),
        "confidence": scoring["confidence"],
        "confidence_normalised": scoring["confidence_normalised"],
        "confidence_display": scoring["confidence_display"],
        "generator_status": "fallback",
        "generator_error": generator_error or "No generation provider returned usable text",
    }

async def generate_appeal_variants(campaign_data: dict, top_n: int = 3) -> list[dict]:
    _ensure_required_generation_fields(campaign_data)

    language = campaign_data.get("language", "English")
    styles = VARIANT_STYLES[:top_n]
    tasks = [
        call_gemini_with_key_fallback(
            _build_generation_prompt(campaign_data, style=style),
            temp=0.5,
            language=language,
            max_retries=1,
            key_offset=index * 3,
        )
        for index, style in enumerate(styles)
    ]
    results = await asyncio.gather(*tasks, return_exceptions=True)

    valid_results: list[tuple[int, dict, str]] = []
    seen_texts: set[str] = set()

    for index, result in enumerate(results):
        if isinstance(result, Exception) or not result or not result.get("appeal_text"):
            logger.warning("Style variant %d failed: %s", index + 1, result)
            continue

        appeal_text = result.get("appeal_text", "").strip()
        if not appeal_text:
            continue

        normalised = re.sub(r"\s+", " ", appeal_text.lower())
        if normalised in seen_texts:
            continue
        seen_texts.add(normalised)
        valid_results.append((index, result, appeal_text))

    scorings = await asyncio.gather(
        *[
            asyncio.to_thread(score_text, appeal_text, language=language)
            for _, _, appeal_text in valid_results
        ]
    )

    candidates: list[dict] = []
    for (index, result, appeal_text), scoring in zip(valid_results, scorings):
        candidates.append({
            "model": result["model"],
            "temperature": 0.5,
            "style": styles[index]["style_name"],
            "appeal_text": appeal_text,
            "quality_label": scoring["quality_label"],
            "final_label": scoring["quality_label"],
            "predicted_class": scoring.get("predicted_class"),
            "class_probs": scoring.get("class_probs", {}),
            "quality_score": scoring["quality_score"],
            "raw_quality_score": scoring.get("raw_quality_score", scoring["quality_score"]),
            "confidence": scoring["confidence"],
            "confidence_normalised": scoring["confidence_normalised"],
            "confidence_display": scoring["confidence_display"],
            "generator_model": result.get("generator_model", result["model"]),
            "generator_status": result.get("generator_status", "live"),
            "generator_error": result.get("generator_error"),
        })

    candidates.sort(key=lambda item: item["quality_score"], reverse=True)
    if not candidates:
        return [_build_fallback_variant(campaign_data)]
    return candidates[:top_n]


class AppealVariantGenerationError(RuntimeError):
    """A later variant failed; callers must not present it as a template card."""

    def __init__(self, message: str, status_code: int = 503):
        super().__init__(message)
        self.status_code = status_code


async def generate_appeal_variant(campaign_data: dict, style_index: int) -> dict:
    """Generate and score one style using the existing cache and key pool."""
    _ensure_required_generation_fields(campaign_data)
    if style_index not in range(len(VARIANT_STYLES)):
        raise ValueError("style_index must be 0, 1, or 2")
    started = time.monotonic()
    language = campaign_data.get("language", "English")
    style = VARIANT_STYLES[style_index]
    result = {}

    async def generate_and_score():
        nonlocal result
        result = await call_gemini_with_key_fallback(
            _build_generation_prompt(campaign_data, style=style),
            temp=0.5, language=language, max_retries=1,
            key_offset=style_index * 3,
        ) or {}
        text = (result.get("appeal_text") or "").strip()
        if not text or result.get("generator_status") == "fallback":
            raise AppealVariantGenerationError(
                result.get("generator_error") or "No generation provider returned usable text"
            )
        scoring = await asyncio.to_thread(score_text, text, language=language)
        return {
            "model": result["model"], "temperature": 0.5,
            "style": style["style_name"], "appeal_text": text,
            "quality_label": scoring["quality_label"],
            "final_label": scoring["quality_label"],
            "predicted_class": scoring.get("predicted_class"),
            "class_probs": scoring.get("class_probs", {}),
            "quality_score": scoring["quality_score"],
            "raw_quality_score": scoring.get("raw_quality_score", scoring["quality_score"]),
            "confidence": scoring["confidence"],
            "confidence_normalised": scoring["confidence_normalised"],
            "confidence_display": scoring["confidence_display"],
            "generator_model": result.get("generator_model", result["model"]),
            "generator_status": result.get("generator_status", "live"),
            "generator_error": result.get("generator_error"),
        }

    try:
        variant = await asyncio.wait_for(generate_and_score(), timeout=VARIANT_TIMEOUT_SECONDS)
    except QualityModelLoadError:
        raise
    except Exception as exc:
        timed_out = isinstance(exc, asyncio.TimeoutError)
        error = (
            f"Variant {style_index + 1} exceeded the {VARIANT_TIMEOUT_SECONDS:g}-second generation ceiling"
            if timed_out else
            str(exc) if isinstance(exc, AppealVariantGenerationError) else
            f"Variant {style_index + 1} generation failed ({type(exc).__name__})"
        )
        if style_index != 0:
            raise AppealVariantGenerationError(error, 504 if timed_out else 503) from exc
        variant = await asyncio.to_thread(
            _build_fallback_variant, campaign_data, error, result.get("generator_model")
        )
    variant["generation_ms"] = max(0, int((time.monotonic() - started) * 1000))
    return variant

async def _legacy_remaining_appeal_variants(campaign_data: dict, skip_styles: int = 1) -> list[dict]:
    _ensure_required_generation_fields(campaign_data)
    results = await asyncio.gather(
        *(generate_appeal_variant(campaign_data, index)
          for index in range(len(VARIANT_STYLES))[skip_styles:]),
        return_exceptions=True,
    )
    candidates = []
    seen_texts = set()
    for result in results:
        if isinstance(result, QualityModelLoadError):
            raise result
        if isinstance(result, Exception):
            logger.warning("Remaining variant failed: %s", type(result).__name__)
            continue
        normalised = re.sub(r"\s+", " ", result["appeal_text"].lower())
        if normalised not in seen_texts:
            seen_texts.add(normalised)
            candidates.append(result)
    return candidates


def _diagnose_weaknesses(appeal_text: str, language: str = "English") -> list[dict]:
    text = (appeal_text or "").strip()
    words = re.findall(r"\S+", text)
    sentences = [s.strip() for s in re.split(r"[.!?।།]+", text) if s.strip()]
    lower_text = text.lower()

    issues: list[dict] = []

    word_count = len(words)
    if word_count < 35:
        issues.append({
            "dimension": "Content depth",
            "severity": "high",
            "issue": "The appeal is too short to explain the need, impact, and donation action clearly.",
            "value": f"{word_count} words",
            "target": "At least 60 words for most channels",
        })
    elif word_count < 60:
        issues.append({
            "dimension": "Content depth",
            "severity": "medium",
            "issue": "The appeal may need a little more detail about what donors will help provide.",
            "value": f"{word_count} words",
            "target": "60-120 words for concise campaign copy",
        })

    donate_terms = (
        "donate", "donation", "give", "support", "contribute", "fund",
        "help provide", "please help", "please support"
    )
    if not any(term in lower_text for term in donate_terms):
        issues.append({
            "dimension": "Call to action",
            "severity": "high",
            "issue": "The appeal does not include a clear donor action.",
            "value": "No explicit ask found",
            "target": "Use a direct request such as donate, support, or contribute",
        })

    impact_terms = (
        "food", "water", "medicine", "medical", "shelter", "clothes", "supplies",
        "hygiene", "families", "households", "children", "elderly"
    )
    if not any(term in lower_text for term in impact_terms):
        issues.append({
            "dimension": "Concrete impact",
            "severity": "medium",
            "issue": "The appeal does not clearly state what support will provide.",
            "value": "Impact terms not detected",
            "target": "Name the supplies, service, or beneficiary group",
        })

    if sentences:
        avg_sentence_len = word_count / len(sentences)
        if avg_sentence_len > 24:
            issues.append({
                "dimension": "Readability",
                "severity": "medium",
                "issue": "Sentences are long, which can make the appeal harder to scan.",
                "value": f"{avg_sentence_len:.1f} words per sentence",
                "target": "10-18 words per sentence",
            })
        elif avg_sentence_len < 6 and word_count >= 20:
            issues.append({
                "dimension": "Sentence structure",
                "severity": "low",
                "issue": "The appeal uses very short fragments and may read less complete.",
                "value": f"{avg_sentence_len:.1f} words per sentence",
                "target": "Complete sentences averaging 10-14 words",
            })

    repeated_words = [
        word
        for word in set(re.findall(r"\b[a-zA-Z]{5,}\b", lower_text))
        if lower_text.count(word) >= 4
    ]
    if repeated_words:
        issues.append({
            "dimension": "Repetition",
            "severity": "low",
            "issue": "Some terms appear repeatedly and may reduce perceived quality.",
            "value": ", ".join(repeated_words[:3]),
            "target": "Avoid repeating key phrases more than needed",
        })

    if not issues:
        issues.append({
            "dimension": "Overall structure",
            "severity": "low",
            "issue": "No major quality issue was detected. A final human review is still recommended.",
            "value": "Stable",
            "target": "Keep the appeal factual, specific, and action-oriented",
        })

    return issues


def _build_improvement_prompt(appeal_text: str, language: str, issues: list[dict]) -> str:
    issue_lines = "\n".join(
        f"- {item['issue']} Target: {item['target']}"
        for item in issues
    )
    word_count = len(appeal_text.split())
    min_words = max(120, word_count * 2)
    return f"""\
TASK: Write a substantially better version of the appeal below.

ORIGINAL APPEAL ({word_count} words):
{appeal_text}

PROBLEMS TO FIX:
{issue_lines}

STRICT REQUIREMENTS FOR THE NEW VERSION:
1. Must be at least {min_words} words. The original is too short.
2. Do not reuse or rearrange sentences from the original. Write fresh.
3. Open with a specific situation statement using the location and need when present.
4. Include a concrete impact statement: exactly what will donations provide?
5. Include one community or human element: who is affected and how.
6. End with a strong, specific call to action, not just "donate today".
7. No invented names, ages, statistics, dates, or details not in the original.
8. CRITICAL: Write every single word — including the call to action — exclusively in {language}.
   Do NOT switch to English for any sentence, phrase, or word. If {language} is Sinhala, the
   call to action must be in Sinhala script. If {language} is Tamil, in Tamil script.
   Any English word in a non-English output is a failure.

Output ONLY the improved appeal. Nothing else.
"""


_FALLBACK_CTA = {
    "Sinhala": "අදම පරිත්‍යාග කරන්න.",
    "Tamil": "இன்றே நன்கொடை வழங்குங்கள்.",
    "English": "Please donate today to provide verified support where it is needed most.",
}

_DONATE_TERMS_BY_LANG: dict[str, tuple[str, ...]] = {
    "Sinhala": ("පරිත්‍යාග", "ආධාර", "දායාද"),
    "Tamil": ("நன்கொடை", "உதவி", "ஆதரவு"),
    "English": ("donate", "support", "contribute", "give"),
}


def _fallback_improved_appeal(appeal_text: str, language: str = "English") -> str:
    text = re.sub(r"\s+", " ", (appeal_text or "").strip())
    if not text:
        return ""
    donate_terms = _DONATE_TERMS_BY_LANG.get(language, _DONATE_TERMS_BY_LANG["English"])
    if any(term in text for term in donate_terms):
        return text
    cta = _FALLBACK_CTA.get(language, _FALLBACK_CTA["English"])
    return f"{text} {cta}"


def _is_weak_improvement(original_text: str, improved_text: str) -> bool:
    original_words = original_text.split()
    improved_words = improved_text.split()
    min_words = max(120, len(original_words) * 2)

    if len(improved_words) < min_words:
        return True

    original_sentences = {
        re.sub(r"\s+", " ", sentence.strip().lower())
        for sentence in re.split(r"[.!?]+", original_text)
        if len(sentence.split()) >= 4
    }
    improved_lower = re.sub(r"\s+", " ", improved_text.lower())
    reused_sentences = sum(
        1
        for sentence in original_sentences
        if sentence and sentence in improved_lower
    )

    return reused_sentences > 0


async def improve_appeal_text(appeal_text: str, language: str = "English") -> dict:
    original = score_text(appeal_text, language)
    issues = _diagnose_weaknesses(appeal_text, language)
    prompt = _build_improvement_prompt(appeal_text, language, issues)

    result = {}
    improved_text = ""

    async def _generate_improvement_with_retry() -> tuple[dict, str]:
        result = await call_gemini_with_key_fallback(prompt, temp=0.35, language=language)
        improved_text = (result.get("appeal_text") or "").strip() if result else ""
        if improved_text and _is_weak_improvement(appeal_text, improved_text):
            retry_prompt = (
                f"{prompt}\n\n"
                "The previous rewrite was too short or too similar. Rewrite again with "
                "new sentence structure, richer factual explanation, and the required "
                "minimum word count. Do not copy original sentences."
            )
            result = await call_gemini_with_key_fallback(retry_prompt, temp=0.45, language=language)
            improved_text = (result.get("appeal_text") or "").strip() if result else ""
        return result or {}, improved_text

    try:
        result, improved_text = await asyncio.wait_for(
            _generate_improvement_with_retry(),
            timeout=15.0,
        )
    except asyncio.TimeoutError:
        logger.warning("Improvement generation timed out — using fallback")

    if not improved_text:
        improved_text = _fallback_improved_appeal(appeal_text, language)

    improved = score_text(improved_text, language)
    return {
        "original_score": original["quality_score"],
        "original_raw_quality_score": original.get("raw_quality_score", original["quality_score"]),
        "original_label": original["quality_label"],
        "original_confidence": original["confidence"],
        "issues": issues,
        "improved_appeal": improved_text,
        "improved_score": improved["quality_score"],
        "improved_raw_quality_score": improved.get("raw_quality_score", improved["quality_score"]),
        "improved_label": improved["quality_label"],
        "improved_confidence": improved["confidence"],
        "method": result.get("model", "fallback_template") if result else "fallback_template",
    }

"""Improve admission policy; never change scores, models, or variant ordering."""
import math

from app.models import appeal_guidance_service as guidance


def _visible_rating(result):
    label = result.get("final_label", result.get("quality_label", result.get("status")))
    rank = guidance.RANK.get(label)
    try:
        score = float(result.get("quality_score", result.get("score")))
    except (TypeError, ValueError):
        return None
    if rank is None or not math.isfinite(score) or not 1 <= score <= 5:
        return None
    # Compare exactly the existing two-decimal UI representation. This does not
    # round or replace the score in the result or in either API response.
    return rank, float(f"{score:.2f}")


def is_visible_improvement(original, candidate):
    """Require a visible gain and retain the existing agreement-first preference."""
    before, after = _visible_rating(original), _visible_rating(candidate)
    if before is None or after is None or after <= before:
        return False
    return guidance.recommended([original, candidate]) is candidate

"""Process-local call budgets, paired-request state, and IP improvement limits."""
from __future__ import annotations

import hashlib
import json
import time
from contextvars import ContextVar
from dataclasses import dataclass, field
from datetime import datetime, timezone
from threading import RLock

from app.models.config import settings

LOCK = RLock()
SESSIONS: dict = {}
DAILY: dict = {}
CURRENT = ContextVar("appeal_call_budget", default=None)


class BudgetExceeded(RuntimeError):
    pass


@dataclass
class Flow:
    attempts: int = 0
    stopped: bool = False


@dataclass
class Budget:
    calls: int = 0
    retry_allowance: int = 0
    started: float = field(default_factory=time.monotonic)
    stopped: bool = False
    limit: int | None = None
    last_model: str | None = None
    last_status: str | None = None
    flow: Flow = field(default_factory=Flow)
    attempts: int = 0

    def remaining_seconds(self):
        return max(0.0, settings.REQUEST_TIME_BUDGET_S - (time.monotonic() - self.started))

    def consume(self):
        with LOCK:
            if self.stopped or self.flow.stopped or self.remaining_seconds() <= 0:
                raise BudgetExceeded("Generation time or provider budget exhausted")
            if self.calls >= (self.limit if self.limit is not None else settings.MAX_GEMINI_CALLS_PER_REQUEST + self.retry_allowance):
                raise BudgetExceeded("Gemini call budget exhausted")
            if self.flow.attempts >= settings.MAX_GEMINI_PROVIDER_ATTEMPTS_PER_FLOW:
                raise BudgetExceeded("Full-flow provider attempt ceiling reached")
            if self.limit == 1 and self.attempts >= 1:
                raise BudgetExceeded("Single-attempt improvement limit reached")
            self.flow.attempts += 1
            self.attempts += 1

    def account(self, billable):
        with LOCK:
            self.calls += int(billable)

    def stop_flow(self):
        self.stopped = self.flow.stopped = True


def normalized(value):
    if isinstance(value, dict):
        return {k: normalized(v) for k, v in sorted(value.items())}
    if isinstance(value, str):
        return " ".join(value.split()).casefold()
    return value


def session(payload, client="local"):
    # A changed provider configuration must not reuse an old provider result.
    key = hashlib.sha256(json.dumps([client, normalized(payload), settings.GEMINI_MODEL,
                                    settings.GEMINI_FALLBACK_MODELS], sort_keys=True).encode()).hexdigest()
    now = time.monotonic()
    with LOCK:
        for old in list(SESSIONS):
            if now - SESSIONS[old]["created"] >= 3600:
                del SESSIONS[old]
        if key not in SESSIONS:
            flow = Flow()
            SESSIONS[key] = {"created": now, "budget": Budget(flow=flow),
                             "remaining_budget": Budget(flow=flow), "flow": flow,
                             "results": {}, "partial": False}
        return SESSIONS[key]


def improvement_day():
    return datetime.now(timezone.utc).date().isoformat()


def refund_improvement(client, day):
    """Release this reservation only; never subtract from a different UTC day."""
    with LOCK:
        key = (day, client)
        if key in DAILY:
            DAILY[key] = max(0, DAILY[key] - 1)
        return max(0, settings.DAILY_IMPROVE_LIMIT - DAILY.get((improvement_day(), client), 0))


def improvement_uses(client, consume=False, day=None):
    day = day or improvement_day()
    with LOCK:
        for key in list(DAILY):
            if key[0] != day:
                del DAILY[key]
        key = (day, client)
        used = DAILY.get(key, 0)
        if consume and used < settings.DAILY_IMPROVE_LIMIT:
            DAILY[key] = used + 1
            used += 1
            allowed = True
        else:
            allowed = not consume
        return allowed, max(0, settings.DAILY_IMPROVE_LIMIT - used)

const MAX_TEXT_LENGTH = 10_000;
const CHAR_WARN_THRESHOLD = 9_000;




class ValidationError extends Error {
  constructor(message) {
    super(message);
    this.name = "ValidationError";
  }
}


function detectScript(text) {
  const sinhalaChars = (text.match(/[\u0D80-\u0DFF]/g) || []).length;
  const tamilChars = (text.match(/[\u0B80-\u0BFF]/g) || []).length;
  const totalChars = text.replace(/\s/g, "").length;
  if (!totalChars) return null;
  if (sinhalaChars / totalChars > 0.3) return "Sinhala";
  if (tamilChars / totalChars > 0.3) return "Tamil";
  return "English";
}


async function analyseAppeal(appealText, language) {
  const trimmed = (appealText || "").trim();
  if (!trimmed) {
    throw new ValidationError("Paste an appeal before analysing.");
  }
  if (trimmed.length < MIN_TEXT_LENGTH) {
    throw new ValidationError(
      `Appeal is too short to analyse (${trimmed.length} characters). Minimum is ${MIN_TEXT_LENGTH}.`
    );
  }
  if (trimmed.length > MAX_TEXT_LENGTH) {
    throw new ValidationError(
      `Appeal exceeds the ${MAX_TEXT_LENGTH.toLocaleString()} character limit.`
    );
  }

  const response = await fetch(`${API_BASE_URL}/api/quality/analyse`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ appeal_text: trimmed, language }),
  });

  if (!response.ok) {
    const data = await response.json().catch(() => null);
    if (response.status === 422) {
      throw new ValidationError(`HTTP ${response.status}: ${data?.detail || "Validation failed."}`);
    }
    throw new Error(`HTTP ${response.status}: ${data?.detail || "Failed to analyse appeal."}`);
  }

  return response.json();
}


async function improveAppeal(appealText, language) {
  const response = await fetch(`${API_BASE_URL}/api/quality/improve`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ appeal_text: appealText, language }),
  });

  if (!response.ok) {
    const data = await response.json().catch(() => null);
    throw new Error(`HTTP ${response.status}: ${data?.detail || "Failed to improve appeal."}`);
  }

  return response.json();
}
